//! Import preserves references protected by immutable native proofs. It creates
//! only zero placeholders for sparse historical slots, never old log payload.
use super::*;
use crate::model::NamespaceId;
use std::collections::{HashMap, HashSet};

#[derive(Default)]
pub(super) struct VectorLayout {
    namespaces: HashMap<u32, (bool, u32)>,
    high: HashMap<NamespaceId, u32>,
    positions: HashSet<(u32, u32)>,
    prepared: bool,
}
impl VectorLayout {
    pub(super) fn observe(&mut self, item: &Item) -> Result<(), StorageError> {
        match item {
            Item::Namespace(ns) => {
                self.namespaces.insert(ns.id.get(), (ns.schema.as_ref().is_some_and(|s| s.vector_certification.is_some()), ns.embedding_dim));
            }
            Item::Memory(memory) => {
                let record = DiskRecord::from_bytes(&memory.record)?;
                let (certified, _) = self.namespaces.get(&record.namespace_id).ok_or_else(|| invalid("import namespace missing"))?;
                if memory.vector_placement.is_some() != (*certified && memory.embedding_bits.is_some()) {
                    return Err(invalid("certified vector placement missing or unmarked"));
                }
                if let Some(slot) = memory.vector_placement {
                    if !self.positions.insert((record.namespace_id, slot)) { return Err(invalid("duplicate protected vector slot")); }
                    if self.positions.len() > crate::storage::vector_certification::MAX_ROWS as usize { return Err(invalid("import protected vector row limit exceeded")); }
                    self.high.entry(NamespaceId::new(record.namespace_id)).and_modify(|n| *n = (*n).max(slot)).or_insert(slot);
                    let bytes = self.high.iter().try_fold(0_u64, |total, (ns, last)| {
                        total.checked_add((u64::from(*last) + 1) * u64::from(self.namespaces[&ns.get()].1) * 4)
                    }).ok_or_else(|| invalid("import protected layout size overflow"))?;
                    if bytes > crate::storage::vector_certification::MAX_BYTES { return Err(invalid("import protected vector layout exceeds 256 MiB")); }
                }
            }
            _ => {}
        }
        Ok(())
    }
    pub(super) fn prepare(&mut self, vectors: &mut crate::storage::VectorManager) -> Result<(), StorageError> {
        if self.prepared { return Ok(()); }
        let high = &self.high;
        let mut bytes = 0_u64;
        for (namespace, slot) in high {
            let store = vectors.get(*namespace).ok_or(StorageError::NamespaceNotFound(namespace.get()))?;
            bytes = bytes.checked_add((u64::from(*slot) + 1) * u64::from(store.dimensions()) * 4)
                .ok_or_else(|| invalid("import physical reference size overflow"))?;
            if bytes > crate::storage::vector_certification::MAX_BYTES {
                return Err(invalid("import protected vector layout exceeds 256 MiB"));
            }
        }
        for (namespace, last) in high {
            let store = vectors.get_mut(*namespace).ok_or(StorageError::NamespaceNotFound(namespace.get()))?;
            let zero = vec![0.; store.dimensions() as usize];
            for slot in 0..=*last {
                if slot >= store.slot_count() || store.is_free(slot) {
                    if store.insert_vector(&zero)? != slot {
                        return Err(invalid("resumed protected vector layout differs"));
                    }
                }
            }
            store.sync()?;
        }
        self.prepared = true;
        Ok(())
    }
    pub(super) fn reclaim_placeholders(meta: &redb::Database, vectors: &mut crate::storage::VectorManager) -> Result<(), StorageError> {
        let read = meta.begin_read()?;
        let mut live = HashSet::new();
        for row in read.open_table(crate::storage::metadata::META_TABLE)?.iter()? {
            let (_, bytes) = row?;
            let record = DiskRecord::from_bytes(bytes.value())?;
            if record.schema.as_ref().and_then(|s| s.vector.as_ref()).is_some() {
                live.insert((record.namespace_id, record.vector_slot));
            }
        }
        let namespaces: Vec<_> = vectors.iter().map(|(id, _)| id).collect();
        for namespace in namespaces {
            let store = vectors.get_mut(namespace).ok_or(StorageError::NamespaceNotFound(namespace.get()))?;
            for slot in 0..store.slot_count() {
                if !store.is_free(slot) && !live.contains(&(namespace.get(), slot)) { store.free_slot(slot)?; }
            }
            store.sync()?;
        }
        Ok(())
    }
}
